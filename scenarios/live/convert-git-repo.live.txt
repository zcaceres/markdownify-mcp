> repomix last-args
expect exact last args: (none)

# GR-9, GR-5 (D3, D5, D11, I3): live GitHub shorthand and explicit branch + compress convert via real repomix CLI
> convert git-repo-to-markdown octocat/Hello-World
expect ok
expect exact ok: git-repo-to-markdown (path: none)
expect contains Hello World!

> convert git-repo-to-markdown https://github.com/octocat/Hello-World --branch=master --compress
expect ok
expect exact ok: git-repo-to-markdown (path: none)
expect contains README

> repomix last-args
expect contains --remote-branch master --compress
