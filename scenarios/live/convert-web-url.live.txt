# Live verification suite: Convert Web URL against real DNS, HTTP, filesystem, and markitdown CLI.
# Run with: bun run harness live scenarios/live/convert-web-url.live.txt

# WU-2 (D14, I1): SSRF guard blocks private IP and file:// before contacting the network
> convert webpage-to-markdown file:///etc/passwd
expect error Only http: and https: schemes are allowed.

> convert webpage-to-markdown http://127.0.0.1/secret
expect error potentially dangerous

> network fetch-count
expect exact fetch count: 0

# WU-3, WU-7, WU-10, WU-14 (D3, D5, I3, I4): live public webpage (Wikipedia) converts via fetch + temp file + real markitdown
> convert webpage-to-markdown https://en.wikipedia.org/wiki/Markdown
expect ok
expect not-empty
expect contains Markdown

> filesystem temp-count
expect exact active temp dirs: 0

# WU-8 (D10, D15): live hosted PDF converts via real markitdown and cleans up temp dir

> filesystem temp-count

# WU-6 (D7, D13): tightening MD_MAX_DOWNLOAD_BYTES rejects a real response that exceeds the cap
> config set MD_MAX_DOWNLOAD_BYTES 50
expect exact config MD_MAX_DOWNLOAD_BYTES = 50

> convert webpage-to-markdown https://www.w3.org/WAI/ER/tests/xhtml/testfiles/resources/pdf/dummy.pdf
expect error Download exceeded maximum size of 50 bytes

> config clear MD_MAX_DOWNLOAD_BYTES
expect exact config cleared MD_MAX_DOWNLOAD_BYTES

> filesystem temp-count
expect exact active temp dirs: 0
