# Slice 01: Server Configuration and HTTP Request Guards (SC-1, SC-2, SC-4, SC-5, SC-6, SC-7)

# SC-1 (D1, D2, D6, D23): defaults select stdio on 127.0.0.1:3000 and CLI flags override MCP_* env vars
> serve config
transport: stdio
host: 127.0.0.1
port: 3000
allowed-hosts: (default)
allowed-origins: (default)
allow-unrestricted-paths: false

> config set MCP_TRANSPORT stdio
config MCP_TRANSPORT = stdio

> config set MCP_PORT 9999
config MCP_PORT = 9999

> config set MCP_ALLOWED_HOSTS mcp.example.com,localhost:4123
config MCP_ALLOWED_HOSTS = mcp.example.com,localhost:4123

> config set MCP_ALLOWED_ORIGINS https://app.example.com
config MCP_ALLOWED_ORIGINS = https://app.example.com

> serve config --transport=http --host=localhost --port=4123 --allow-unrestricted-paths
transport: http
host: localhost
port: 4123
allowed-hosts: mcp.example.com, localhost:4123
allowed-origins: https://app.example.com
allow-unrestricted-paths: true

# SC-5, SC-6 (D33, I5): non-loopback bind allows valid Host unless MCP_ALLOWED_HOSTS set, and blocks Origins unless listed
> config set MCP_HOST 0.0.0.0
config MCP_HOST = 0.0.0.0

> serve check-http POST /mcp host=mcp.example.com
http: allow

> serve check-http POST /mcp host=mcp.example.com origin=https://app.example.com
http: 403 Forbidden: invalid Origin header

> config set MCP_ALLOWED_HOSTS mcp.example.com
config MCP_ALLOWED_HOSTS = mcp.example.com

> config set MCP_ALLOWED_ORIGINS https://app.example.com
config MCP_ALLOWED_ORIGINS = https://app.example.com

> serve check-http POST /mcp host=other.example.com
http: 403 Forbidden: invalid Host header

> serve check-http POST /mcp host=mcp.example.com origin=https://app.example.com
http: allow
